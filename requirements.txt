# ----- PDF 文字提取（主力：PyMuPDF）-----
# 实际测试对中文 PDF 字符识别 100%，比 pdfplumber / MarkItDown 更稳定
PyMuPDF>=1.23.0

# ----- OCR 主力：RapidOCR（v3.3 升级）-----
# 基于 PaddleOCR 模型 + ONNX Runtime 推理，无需安装 PaddlePaddle 框架
# pip 一条命令安装，跨平台，中文手写体 85%+，表格识别输出 HTML 结构
# v3.3 调优参数：det_limit_side_len=1920, det_db_thresh=0.22,
#               多尺度/多预处理重试，行聚类后处理
rapidocr-onnxruntime>=1.3.0

# ----- PDF 转图片（pdf2image + poppler）-----
# pdf2image 依赖 poppler（PDF 拆图用）
# Poppler 由 install.ps1 自动下载，不打包在 Skill 中
pdf2image>=1.16.3

# Pillow 是 RapidOCR / PaddleOCR / pdf2image 共同依赖
Pillow>=10.0.0

# ----- docx 读取（如需审核 .docx 资料）-----
python-docx>=1.1.0

# =============================================================
# 可选依赖（增强能力）
# =============================================================
# PaddleOCR — 显式备选 OCR 引擎（--engine paddle）
# install.ps1 会自动安装，但默认不启用
# Windows 实测 3.x 有 oneDNN 兼容问题，固定 2.8.1 + PaddlePaddle 2.6.2
paddleocr==2.8.1
paddlepaddle==2.6.2
opencv-python>=4.8.0

# rapid-table — 表格结构检测（--use-table）
# 对密集表格行丢失问题最有效，install.ps1 会自动安装
rapid-table

# Tesseract — 当 RapidOCR/PaddleOCR 不可用时降级使用
# 注意：还需要单独安装 Tesseract 引擎本体（系统级）
# Windows: https://github.com/UB-Mannheim/tesseract/wiki
#   安装时勾选 "Chinese (Simplified)" 和 "English" 语言包
# install.ps1 会自动下载安装 Tesseract + 中文语言包
pytesseract>=0.3.10
